This example is a Container that holds one integer, and a ContainerProcess that adds its argument to that value.

## Container

<Tabs syncKey="lang">
<TabItem label="Rust">

```rust
use juiz_sdk::prelude::*;

#[repr(Rust)]
pub struct ExampleContainer {
    pub value: i64
}

#[juiz_container]
fn example_container(initial_value: i64) -> JuizResult<Box<ExampleContainer>> {
    Ok(Box::new(ExampleContainer { value: initial_value }))
}
```

</TabItem>
<TabItem label="C++">

```cpp
// example_container.h
#pragma once
#include <cstdint>

class CppContainer {
public:
    int64_t value;
    CppContainer(int64_t v) : value(v) {}
};
```

```cpp
// example_container_cpp.cpp
#include "juiz/juiz.h"
#include "example_container.h"

juiz::Value manifest() {
    return ContainerManifest("example_container_cpp").into_value();
}

CppContainer* create_container(juiz::Value value) {
    int64_t int_value = 0;
    if (value.isObjectValue() && value.hasKey("value")) {
        auto v = value.objectValue()["value"];
        if (v.isIntValue()) int_value = v.intValue();
    }
    return new CppContainer(int_value);
}

bool destroy_container(CppContainer* p_container) {
    if (p_container) {
        delete p_container;
        return true;
    }
    return false;
}

CONTAINER_FACTORY(manifest, create_container, destroy_container);
```

</TabItem>

<TabItem label="Haskell">

```haskell
-- ExampleContainerState.hs: the state type (put it in both the Container and the ContainerProcess libraries)
module ExampleContainerState (ExampleContainer (..)) where

import Data.IORef (IORef)
import Data.Int (Int64)

newtype ExampleContainer = ExampleContainer (IORef Int64)
```

```haskell
-- ExampleContainer.hs
{-# LANGUAGE OverloadedStrings #-}
{-# LANGUAGE TemplateHaskell #-}
module ExampleContainer () where

import Juiz
import ExampleContainerState

manifest :: ContainerManifest
manifest = containerManifest "example_container_hs"
  & addIntArg "value" "initial value" 0

create :: Int64 -> IO ExampleContainer
create value = ExampleContainer <$> newIORef value

juizContainer 'create 'manifest
```

The arguments of `containerManifest` become the arguments of the function that creates the Container (it receives the values written under `containers` in the configuration file). Keep values you rewrite in an `IORef`.

</TabItem>
<TabItem label="Zig">

```zig
// example_container_state.zig: the state type (both the Container and the ContainerProcess libraries @import it)
pub const Counter = struct {
    value: i64,
};
```

```zig
// example_container_zig.zig
const juiz = @import("juiz");
const state = @import("example_container_state.zig");

fn create(args: struct { value: i64 = 0 }) state.Counter {
    return .{ .value = args.value };
}

comptime {
    juiz.exportContainer(create, .{
        .type_name = "example_container_zig",
        .args = .{ .value = "initial value" },
    });
}
```

The function that makes the Container takes its arguments as one struct, like a Process (it gets the values written in `containers` of the configuration file). zigjuiz keeps the returned value as the state of the Container. If the state type has `pub fn deinit(self: *State) void`, it is called when the Container is removed.

</TabItem>
</Tabs>

## ContainerProcess

A ContainerProcess is a Process that receives the Container as its first argument. `container_type` gives the type name of the Container it is bound to.

<Tabs syncKey="lang">
<TabItem label="Rust">

```rust
use example_container::ExampleContainer;
use juiz_sdk::prelude::*;

#[juiz_container_process(container_type = "example_container")]
fn increment_function(container: &mut ContainerImpl<ExampleContainer>, arg1: i64) -> JuizResult<Capsule> {
    container.value = container.value + arg1;
    return Ok(jvalue!(container.value).into());
}
```

</TabItem>
<TabItem label="C++">

```cpp
#include "juiz/juiz.h"
#include "example_container.h"

juiz::Value manifest() {
    return ProcessManifest("example_container_cpp_increment")
        .container_type("example_container_cpp")
        .add_int_arg("arg0", "test_argument", 2)
        .into_value();
}

std::optional<int64_t> example_container_increment(CppContainer* container, juiz::CapsuleMap cm) {
    int64_t v = cm.get_int("arg0");
    container->value = container->value + v;
    return container->value;
}

CONTAINER_PROCESS_FACTORY(CppContainer, manifest, example_container_increment)
```

</TabItem>
<TabItem label="Python">

```python
from juiz import juiz_container_process

@juiz_container_process(container_type="example_container_python")
def example_container_python_get(container) -> int:
    return container.value
```

</TabItem>
<TabItem label="Haskell">

```haskell
{-# LANGUAGE OverloadedStrings #-}
{-# LANGUAGE TemplateHaskell #-}
module ExampleContainerIncrement () where

import Juiz
import ExampleContainerState

manifest :: ProcessManifest
manifest = containerProcessManifest "example_container_hs_increment" "example_container_hs"
  & addIntArg "arg1" "value to be added" 1

increment :: ExampleContainer -> Int64 -> IO Int64
increment (ExampleContainer ref) arg1 = atomicModifyIORef' ref (\v -> (v + arg1, v + arg1))

juizContainerProcess 'increment 'manifest
```

The second argument of `containerProcessManifest` is the type name of the Container to bind to.
Use the same type from the same module for the state in both the Container library and the ContainerProcess library (a different type causes an error). To put them in one library, use `juizComponent` ([tutorial chapter 6 (Haskell)](/tutorial/component/haskell/)).

</TabItem>
<TabItem label="Zig">

```zig
// example_container_zig_increment.zig
const std = @import("std");
const juiz = @import("juiz");
const state = @import("example_container_state.zig");

fn increment(self: *state.Counter, args: struct { arg1: i64 = 1 }) !i64 {
    self.value = try std.math.add(i64, self.value, args.arg1);
    return self.value;
}

comptime {
    juiz.exportContainerProcess(increment, .{
        .type_name = "example_container_zig_increment",
        .container_type = "example_container_zig",
        .args = .{ .arg1 = "value to be added" },
    });
}
```

The first argument `self` receives the state of the Container. `.container_type` is the type name of the Container to bind to. Taking `self` as `*const` makes the ContainerProcess readonly.
Use the same type from the same file (`@import`) in the libraries of the Container and the ContainerProcess (if the types differ, the function is not called and -1104 is returned). To put them in one library, use `juiz.exportComponent` ([Tutorial chapter 6 (Zig)](/tutorial/component/zig/)).

</TabItem>

## Running it on its own

<Tabs syncKey="platform">
<TabItem label="Linux">

```sh
juiz --container ./target/debug/libexample_container.so \
     --container_process ./target/debug/libexample_container_increment.so \
     -l rust -e -1 -d
```

</TabItem>
<TabItem label="macOS">

```sh
juiz --container ./target/debug/libexample_container.dylib \
     --container_process ./target/debug/libexample_container_increment.dylib \
     -l rust -e -1 -d
```

</TabItem>
<TabItem label="Windows">

```powershell
juiz --container ./target/debug/example_container.dll `
     --container_process ./target/debug/example_container_increment.dll `
     -l rust -e -1 -d
```

On Windows, the file names do not start with `lib`.
